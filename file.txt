BEGIN
    read x
    IF (x < 5) THEN
        write "x < 5"
        write "\nmiao"
    ELIF (x == 5) THEN
        REPEAT
            read x
            IF (x == "asd") THEN
                y[BEGIN] <- ""
                y <- y + 1
                y <- y + 2
                y <- y + 3
                i <- 0
                WHILE (i < y?)
                BEGIN
                    write y[i]
                    i <- i + 1
                END
            ENDIF
        UNTIL (x != "ciao")
        write "fatto"
    ELIF (x == 6) THEN
        write x, "\n"
    ELSE
        write "x > 6"
        write "\nciao"
    ENDIF
    write "\nend"
END
